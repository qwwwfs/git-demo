cmake_minimum_required(VERSION 3.16)
project(PidBasics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pid_controller pid_controller.cpp)
target_include_directories(pid_controller PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(pid_step_demo main.cpp)
target_link_libraries(pid_step_demo PRIVATE pid_controller)
add_executable(pid_tests tests.cpp)
target_link_libraries(pid_tests PRIVATE pid_controller)
enable_testing()
add_test(NAME pid_unit_tests COMMAND pid_tests)
